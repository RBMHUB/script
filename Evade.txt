            ringStroke.Color = THEME.danger
            CenterIcon.Text = "!"
            CenterIcon.TextColor3 = THEME.danger
            LoadingStatus.TextColor3 = THEME.danger
            PercentLabel.Visible = false
            ProgressBG.Visible = false
            CloseButton.Visible = true
        elseif state == "success" then
            ringStroke.Color = THEME.success
            CenterIcon.Text = "✓"
            CenterIcon.TextColor3 = THEME.success
            LoadingStatus.TextColor3 = THEME.text
            PercentLabel.TextColor3 = THEME.success

            ringStroke.Color = THEME.primary
            CenterIcon.Text = ""
            CenterIcon.TextColor3 = THEME.primary
            LoadingStatus.TextColor3 = THEME.text

            BackgroundTransparency = 1,

        fadeTween.Completed:Connect(function()
            ScreenGui:Destroy()
        end)

    )
    local success, result = pcall(function()
        return game:HttpGet(url .. cacheBuster, true)
    end)
    return success and result or nil
end

local function main()
    local ui = createLoaderUI()
    task.wait(0.2)

    local currentPlaceId = game.PlaceId
    local selectedGame = nil
    local gameData = nil

    for gameName, data in pairs(SUPPORTED_GAMES) do
        for _, placeId in ipairs(data.placeIds) do
            if currentPlaceId == placeId then
                selectedGame = gameName
                gameData = data
                break
            end
        end
        if selectedGame then break end
    end

    if not selectedGame then
        ui.setGameBadge("Unsupported", "error")
        ui.showLoading()
        ui.setStatus("Game Not Supported", "PlaceId (" .. tostring(currentPlaceId) .. ") is not in the supported list.",
            "error")
        return
    end

    ui.setGameBadge(selectedGame, "ready")

    local hasFarmOnly = gameData.farmOnly ~= nil
    local hasVisualOnly = gameData.visualOnly ~= nil

    if not hasFarmOnly then
        ui.TpCard.Visible = false
        ui.FullCard.Position = UDim2.new(0, 0, 0, 0)
        ui.FullTitle.Text = "Launch Script"
        ui.FullDesc.Text = "Execute the main script for " .. selectedGame
    end

    if hasVisualOnly then
        ui.VisualCard.Visible = true
        ui.VisualCard.Position = UDim2.new(0, 0, 0, hasFarmOnly and (ui.RowStep * 2) or ui.RowStep)
    end
    ui.layout(true)

    local isLaunching = false
    local function loadScript(version)
        if isLaunching or not gameData then return end
        local scriptUrl = gameData[version]
        if not scriptUrl then return end
        isLaunching = true

        ui.showLoading()
        ui.setStatus("Fetching script...", "Connecting to delivery server")
        ui.updateProgress(0.3)
        task.wait(0.25)

        local scriptContent = fetchScript(scriptUrl)

        if not scriptContent then
            ui.setStatus("Connection Failed", "Could not reach script server. Check your network.", "error")
            ui.updateProgress(0)
            task.wait(3.5)
            ui.close()
            return
        end

        local loadHint = "Loading Visual Only module..."
        if version == "full" then
            loadHint = "Loading Full Feature module..."
        elseif version == "farmOnly" then
            loadHint = "Loading Farm Only module..."
        end
        ui.setStatus("Executing " .. selectedGame, loadHint)
        ui.updateProgress(0.75)
        task.wait(0.25)

        local success, err = pcall(function()
            getgenv().SCRIPT_KEY = "KEYLESS"
            loadstring(scriptContent)()
        end)

        if success then
            ui.setStatus("Executed Successfully!", "Enjoy Bagah Hub", "success")
            ui.updateProgress(1)
            task.wait(1.1)
            ui.close()
        else
            ui.setStatus("Execution Error", tostring(err), "error")
            ui.updateProgress(0)
            task.wait(4.5)
            ui.close()
        end
    end

    ui.FullCard.MouseButton1Click:Connect(function()
        loadScript("full")
    end)

    ui.TpCard.MouseButton1Click:Connect(function()
        if hasFarmOnly then
            loadScript("farmOnly")
        end
    end)

    ui.VisualCard.MouseButton1Click:Connect(function()
        if hasVisualOnly then
            loadScript("visualOnly")
        end
    end)
end

main()
