local function cueDeflect(ae,af,ag,ah)
if not(ag and ag.Kind=="Ball"and ag.ObjectDirection)then return nil,nil end
local ai=ag.CueDirection
local E=ag.CueDistance
if ai and E then return ai,E end
local F=i.Physics
if not F then return nil,nil end
if not ai then
local G=af:Dot(ag.ObjectDirection)
local H=af-ag.ObjectDirection*(G*(1+(F.BallRestitution or 0.95))/2)
if H.Magnitude>0.05 then ai=H.Unit end
end
if not E then
local G,H=pcall(function()return k.GetShotSpeed(ae,ah)end)
local I=k.RollDistance~=nil
if G and I and H then
local J=math.sqrt(math.max(H*H-2*(F.RollingDeceleration or 20)*(ag.Distance or 0),0))
local K=math.clamp(ag.ImpactFraction or 0,0,0.999)
local L,M=pcall(k.RollDistance,J*math.sqrt(1-K*K))
if L then E=M end
end
end
return ai,E
end

local function aimAssistStep(E)
if not r.AimAssist or r.StickyAim or ab.aimAnimating()or ae then
if ah then ah=false;ab.releaseAim()end;return
end
local F=u.Input
if not(F and F.State=="Aiming")or not isMyTurn()then
if ah then ah=false;ab.releaseAim()end;return
end
local G=F.Direction
if not G or G.Magnitude<1e-4 then return end
G=G.Unit
local H=B.best or B.compute(false)
if not(H and H.Dir)then return end
local I=math.clamp(G:Dot(H.Dir),-1,1)
local J=math.acos(I)
if J>math.rad(12)then if ah then ah=false;ab.releaseAim()end;return end
local K=d:GetMouseLocation()
local L=ai and(K-ai).Magnitude or 0
ai=K
if L>4 then if ah then ah=false;ab.releaseAim()end;return end
local M=G.X*H.Dir.Y-G.Y*H.Dir.X
local N=M>=0 and 1 or-1
if J<0.005 then F.Direction=H.Dir
else
local O=math.clamp(J*E*7,0.002,J)
local P=math.atan2(G.Y,G.X)+N*O
F.Direction=Vector2.new(math.cos(P),math.sin(P))
end
F.AimLocked=true;ah=true
end

local function aimAnimStep()
if not aa then return end
local E=u.Input
if not E then aa=nil;return end
local F=aa
local G=math.clamp((os.clock()-F.t0)/F.dur,0,1)
local H=1-(1-G)^3
local I=F.total*H
E.Direction=Vector2.new(math.cos(F.base+F.turn*I),math.sin(F.base+F.turn*I))
if G>=1 then
E.Direction=Vector2.new(math.cos(F.base+F.turn*F.total),math.sin(F.base+F.turn*F.total))
aa=nil
end
end

local function automationStep()

if not E then return end
local F=E.State
local G,H,I=isMyTurn()

local J=H and(tostring(H.ShotIndex)..":"..tostring(H.Turn))or nil
if J~=ag then
ag=J
af=false
if H and H.Turn==I then
task.defer(function()B.compute(true)end)
end
end


if F=="BallInHand"and r.AutoPlaceBIH and not af and G then
af=true
task.spawn(function()
task.wait(0.2)
if u.Input and u.Input.State=="BallInHand"then
ab.placeBest()
task.wait(0.1)
B.bestKey=nil

end
end)
end

if r.AutoPlay and G and not ae and(F=="Aiming"or(F=="BallInHand"and af))then

task.wait(math.max(r.AutoDelay or 0.6,0.25))
local L=u.Input
local M=L and L.State

ae=false;return
end
local N=B.compute(true)
if N then

if N.Ball and N.Ball.Number==8 then

end
